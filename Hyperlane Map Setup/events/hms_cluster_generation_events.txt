system_event = {
	id = hms_hubs.2
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_spawn_system_batch = begin
		log = "Stage [This.cluster_connection_range] Cluster Connections"
		event_target:hms_hyperlane_hub_primary = {
			change_variable = {
				which = cluster_connection_range
				value = 1
			}
		}
		every_system = {
			limit = {
				has_star_flag = hms_hyperlane_hub
			}
			every_neighbor_system_euclidean = {
				limit = {
					NOR = {
						has_star_flag = hms_hyperlane_hub
						has_star_flag = sealed_system
						has_star_flag = hms_hyperlane_hub_spread
						has_hyperlane_to = prev
					}
				}
				add_hyperlane = {
					from = this
					to = prev
				}
				set_star_flag = hms_hyperlane_hub_spread
				set_star_flag = hms_hyperlane_cluster@root
			}
		}
		set_spawn_system_batch = end
		system_event = {
			id = hms_hubs.3
		}
	}
	after = {
	}
}

system_event = {
	id = hms_hubs.3
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				any_system = {
					hms_isolated_system = no
					count_neighbor_system = {
						count = 0
					}
				}
				check_variable = {
					which = cluster_connection_range
					value <= 100
				}
			}
			log = "Stage [This.cluster_connection_range] Cluster Connections"
			change_variable = {
				which = cluster_connection_range
				value = 1
			}
			set_spawn_system_batch = begin
			every_system = {
				limit = {
					has_star_flag = hms_hyperlane_hub_spread
				}
				every_neighbor_system_euclidean = {
					limit = {
						NOR = {
							has_star_flag = hms_hyperlane_hub
							has_star_flag = sealed_system
							has_star_flag = hms_hyperlane_hub_spread
							has_hyperlane_to = prev
						}
					}
					add_hyperlane = {
						from = this
						to = prev
					}
					set_star_flag = hms_hyperlane_hub_spread
				}
			}
			set_spawn_system_batch = end
			system_event = {
				id = hms_hubs.3
			}
		}
		else = {
			system_event = {
				id = hms_hubs.4
			}
		}
	}
}

system_event = {
	id = hms_hubs.4
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		log = "Conneting Cluster Systems"
		set_spawn_system_batch = begin
		switch = {
			trigger = has_global_flag
			hms_cluster_hyperlanes_neighbour = {
				every_system = {
					limit = {
						has_star_flag = hms_hyperlane_hub_spread
					}
					every_neighbor_system_euclidean = {
						limit = {
							NOT = {
								has_hyperlane_to = prev
							}
							OR = {
								has_star_flag = hms_hyperlane_hub
								has_star_flag = hms_hyperlane_hub_spread
							}
							NOT = {
								has_star_flag = sealed_system
							}
							distance = {
								source = prev
								min_jumps = 1
								max_jumps = 10
								type = hyperlane
							}
							distance = {
								source = prev
								min_distance >= 1
								max_distance <= 40
								type = euclidean
							}
						}
						add_hyperlane = {
							from = this
							to = prev
						}
					}
				}
			}
			hms_cluster_hyperlanes_hub = {
				every_system = {
					limit = {
						has_star_flag = hms_hyperlane_hub_spread
					}
					every_system = {
						limit = {
							has_star_flag = hms_hyperlane_hub
							distance = {
								source = prev
								min_jumps = 1
								max_jumps = 10
								type = hyperlane
							}
						}
						add_hyperlane = {
							from = this
							to = prev
						}
					}
					every_system = {
						limit = {
							has_star_flag = hms_hyperlane_hub_spread
							distance = {
								source = prev
								min_jumps = 1
								max_jumps = 10
								type = hyperlane
							}
						}
						remove_hyperlane = {
							from = this
							to = prev
						}
					}
				}
			}
		}
		set_spawn_system_batch = end
		system_event = {
			id = hms_hubs.5
		}
	}
}

system_event = {
	id = hms_hubs.5
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		log = "Link Clusters"
		set_spawn_system_batch = begin
		hms_utilities_set_hub_stars = yes
		switch = {
			trigger = has_global_flag
			hms_hub_hyperlanes = {
				every_system = {
					limit = {
						has_star_flag = hms_hyperlane_hub
					}
					every_system = {
						limit = {
							has_star_flag = hms_hyperlane_hub
							NOT = {
								has_hyperlane_to = prev
							}
							distance = {
								source = prev
								min_distance >= 1
								max_distance <= 170
								type = euclidean
							}
						}
						add_hyperlane = {
							from = this
							to = prev
						}
					}
				}
			}
			hms_hub_wormholes = {
				every_system = {
					limit = {
						has_star_flag = hms_hyperlane_hub
					}
					set_variable = { which = hms_wormhole_fails value = 0 }
					system_event = { id = hms_hubs.6 }
				}
			}
			hms_hub_gateways = {
				every_system = {
					limit = {
						has_star_flag = hms_hyperlane_hub
					}
					spawn_megastructure = {
						type = "hms_gateway"
						orbit_distance = 20
						random_pos = yes
						orbit_angle = 360						# range of the randomization
					}
				}
			}
			hms_standard_hyperlanes = {
				every_system = {
					limit = {
						has_star_flag = hms_hyperlane_hub_spread
					}
					every_neighbor_system_euclidean = {
						limit = {
							NOR = {
								has_hyperlane_to = prev
								distance = {
									source = prev
									min_jumps = 1
									max_jumps = 10
									type = hyperlane
								}
							}
						}
						#Something about connecting the clusters through normal hyperlanes.
					}
				}
			}
		}
		set_spawn_system_batch = end
	}
}

system_event = {
	id = hms_hubs.6
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		log = "Making Wormhole connections for Cluster [this.GetName]"
		while = {
			limit = {
				has_star_flag = hms_hyperlane_hub
				check_variable = { which = hms_wormhole_fails value < 4 }
				any_system = {
					has_star_flag = hms_hyperlane_hub
					distance = {
						source = prev
						min_distance >= 1
						max_distance <= 170
						type = euclidean
					}
					NOT = { has_star_flag = hms_wormhole_connected@prev }
					NOT = { prev = { has_star_flag = hms_wormhole_connected@prev } }
					NOT = { is_same_value = prev }
				}
			}
			log = "Wormhole connection fails for Cluster [this.GetName]: [this.hms_wormhole_fails]"
			save_event_target_as = hms_hub_wormhole_source
			change_variable = { which = hms_wormhole_fails value = 1 }
			#Check if this system has a wormhole
			if = {
				limit = {
					has_natural_wormhole = no
				}
				system_event = { id = hms_hubs.7 }
			}
			#If it does we have to put it in a different system in the cluster
			else = {
				random_system = {
					limit = {
						has_natural_wormhole = no
						NOT = { has_star_flag = hms_hyperlane_hub }
						distance = {
							source = prev
							min_jumps = 1
							max_jumps = 20
							type = hyperlane
						}
					}
					system_event = { id = hms_hubs.7 }					
				}
			}
		}
	}
}
