namespace = BiggerColonies

event = {
	id = BiggerColonies.0
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_global_flag = bigger_colonies_defines_active
		every_galaxy_planet = {
			limit = {
				colonizable_planet = yes
				is_artificial = no
				NOT = { has_planet_flag = bigger_colony_checked }
			}
			planet_event = { id = BiggerColonies.1 days = 1 }
		}
	}
}

planet_event = {
	id = BiggerColonies.1
	hide_window = yes
    is_triggered_only = yes
	trigger = {
		colonizable_planet = yes
		NOT = { has_planet_flag = bigger_colony_checked }
	}
    immediate = {
		set_planet_flag = bigger_colony_checked
		while = { #Add as many random deposits as there is half the planet size
			count = trigger:planet_size
			random_list = {
				1 = {} 
				1 = { 
					random_list = { 1 = {} 2 = { bc_add_random_deposit = yes } }
					random_list = { 1 = {} 1 = { bc_add_random_blocker = yes } }
					random_list = { 1 = {} 1 = { bc_add_random_diversity_mod_deposit = yes } } # !!District Diversity Mod
				}						
			} 
		}
	}
}
