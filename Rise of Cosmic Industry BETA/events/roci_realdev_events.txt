namespace = realdev

country_event = {
	id = realdev.0
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = no_realdev }
		OR = {
			is_country_type = default
			is_country_type = fallen_empire
		}
	}
	
	immediate = {
		reset_realdev_variables = yes
		every_owned_leader = {
			leader_event = { id = realdev.71 }
		}
		if = {
			limit = {
				NOT = { has_country_flag = realdev_variables_set }
			}
			set_realdev_country_variables = yes
		}
		every_owned_colony = {
			reset_realdev_variables = yes
			#Start of game stuff to immediately apply the modifiers
			colony_event = { id = realdev.1 }
			colony_event = { id = realdev.2 }
			colony_event = { id = realdev.3 }
		}
		
		#Country start of game stuff after planets have figured out their stuff.
		country_event = { id = realdev.4 days = 1 }
		country_event = { id = realdev.5 days = 2 }
		
		every_owned_colony = {
			reset_realdev_variables = yes
			colony_event = { id = realdev.1 days = 1 random = 6 }
			colony_event = { id = realdev.2 days = 8 random = 6 }
			colony_event = { id = realdev.3 days = 14 random = 6 }
		}
		country_event = { id = realdev.4 days = 21 random = 8 }
		#Export trade value for all countries now so it can be shared later
		export_trigger_value_to_variable = {
			trigger = resource_revenue_compare
			parameters = { resource = trade }
			variable = realdev_trade_value
		}
		country_event = { id = realdev.5 days = 30 }
	}
}

#First event for planet math
colony_event = {
	id = realdev.1
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = no_realdev }
	}
	
	immediate = {
		#Include Building Production, Branch Offices too
		
		set_variable = { which = realdev_construction_ship_industry value = 0 }
		set_variable = { which = realdev_alloy_production value = 0 }
		set_variable = { which = realdev_congoods_production value = 0 }
		set_variable = { which = realdev_building value = 0 }
		set_variable = { which = realdev_industry_pops value = 0 }
		
		set_variable = { which = realdev_building value = modifier:realdev_ip_mod }
		
		#Include a bonus for capital planets
		if = {
			limit = {
				is_capital = yes
			}
			change_variable = { which = realdev_building value = 20 }
		}
		#Include base capacity based on Capital Buildings
		if = {
			limit = {
				has_fully_upgraded_capital = yes
			}
			change_variable = { which = realdev_building value = 30 }
		}
		else_if = {
			limit = {
				has_major_upgraded_capital = yes
			}
			change_variable = { which = realdev_building value = 20 }
		
		}
		else_if = {
			limit = {
				has_upgraded_capital = yes
			}
			change_variable = { which = realdev_building value = 10 }
		
		}
		else = {
			change_variable = { which = realdev_building value = 5 }
		}
		
		if = {
			limit = {
				log_realdev = yes
			}
			log = "Realdev industry [this.realdev_industry] before adding buildings [this.realdev_building] on [This.GetName] for [Owner.GetName]"
		}
		
		change_variable = { which = realdev_industry value = realdev_building }
		
		if = {
			limit = {
				log_realdev = yes
			}
			log = "Realdev industry [this.realdev_industry] after adding buildings on [This.GetName] for [Owner.GetName]"
		}
		export_trigger_value_to_variable = {
			trigger = count_owned_workforce
			variable = realdev_industry_pops
			rounded = yes
		}
		
		if = {
			limit = {
				log_realdev = yes
			}
			log = "Realdev industry [this.realdev_industry] before adding workforce [this.realdev_industry_pops] on [This.GetName] for [Owner.GetName]"
		}
		change_variable = { which = realdev_industry value = realdev_industry_pops }
		
		if = {
			limit = {
				log_realdev = yes
			}
			log = "Realdev industry [this.realdev_industry] after adding pops on [This.GetName] for [Owner.GetName]"
		}
	}
}

#Third event for planet math
colony_event = {
	id = realdev.3
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = no_realdev }
	}
	
	immediate = {
		#Include bonuses from Capital Bonus techs, Governor, Ruler, and Amenities
		set_variable = { which = realdev_industry_multiplier_bonus value = 100 }
		set_variable = { which = realdev_industry_multiplier_bonus_2 value = 1 }
		change_variable = { which = realdev_industry_multiplier_bonus value = value:realdev_industry_multiplier }
		
		export_trigger_value_to_variable = { trigger = free_amenities variable = realdev_amenities }
		if = {
			limit = {
				check_variable = { which = realdev_amenities value > 10000 }
			}
			set_variable = { which = realdev_amenities value = 10000 }
		}
		else_if = {
			limit = {
				check_variable = { which = realdev_amenities value < -10000 }
			}
			set_variable = { which = realdev_amenities value = -10000 }
		}
		divide_variable = { which = realdev_amenities value = 10000 }
		change_variable = { which = realdev_industry_multiplier_bonus_2 value = realdev_amenities }
		
		#Apply Multiplier Bonuses
		divide_variable = { which = realdev_industry_multiplier_bonus value = 100 }
		
		multiply_variable = { which = realdev_alloy_production value = prealdev_industry_multiplier_bonus }
		multiply_variable = { which = realdev_congoods_production value = realdev_industry_multiplier_bonus }
		multiply_variable = { which = realdev_building value = realdev_industry_multiplier_bonus }
		multiply_variable = { which = realdev_industry_pops value = realdev_industry_multiplier_bonus }
		
		multiply_variable = { which = realdev_alloy_production value = prealdev_industry_multiplier_bonus_2 }
		multiply_variable = { which = realdev_congoods_production value = realdev_industry_multiplier_bonus_2 }
		multiply_variable = { which = realdev_building value = realdev_industry_multiplier_bonus_2 }
		multiply_variable = { which = realdev_industry_pops value = realdev_industry_multiplier_bonus_2 }
		
		#Include Bonus from orbiting Construction Ships
		set_variable = { which = realdev_construction_ship_industry value = 0 }
		if = {
			limit = {
				carrier = {
					count_fleet_in_orbit = {
						count > 0
						limit = { any_owned_ship = { is_ship_size = constructor } }
					}
				}
			}
			#5% of everything else first
			change_variable = { which = realdev_construction_ship_industry value = realdev_alloy_production }
			change_variable = { which = realdev_construction_ship_industry value = realdev_congoods_production }
			change_variable = { which = realdev_construction_ship_industry value = realdev_building }
			change_variable = { which = realdev_construction_ship_industry value = realdev_industry_pops }
			
			multiply_variable = { which = realdev_construction_ship_industry value = 0.05 }
			
			change_variable = { which = realdev_construction_ship_industry value = 20 }
		}
	}
}

		round_variable = realdev_alloy_production

		change_variable = { which = realdev_total_ip value = realdev_building }

		set_variable = { which = realdev_pb_divisor value = 1 }
		add_variable = { which = realdev_pb_divisor value = realdev_production_boost_perm_display }
		set_variable = { which = realdev_pb_stripper value = 1 }
		divide_variable = { which = realdev_pb_stripper value = realdev_pb_divisor }
		multiply_variable = { which = realdev_industry_pops value = realdev_pb_stripper }

		round_variable = realdev_industry_pops
		round_variable = realdev_trade_value_total

		change_variable = { which = realdev_total_ip value = realdev_industry_pops }
		change_variable = { which = realdev_total_ip value = realdev_trade_value_total }
		
		set_variable = { which = realdev_industry_display value = realdev_total_ip }
		#Calculate how much IP each thing needs
		withdraw_ip = {
			ip_target = realdev_planet_build_speed
			ip_withdrawn = value:realdev_planet_target_calc|TARGET_VARIABLE|realdev_planet_build_speed|CONVERSION_RATIO|@
		}

			ip_withdrawn = value:realdev_country_target_calc|TARGET_VARIABLE|realdev_ship_build_speed|MULT_VARIABLE|realdev_shipyard_count|

						is_robotic = yes

						has_trait = trait_clone_soldier_infertile
						has_trait = trait_clone_soldier_infertile_full_potential
						AND = {
							is_organic_species = yes
							PREV = { is_hive_empire = yes }
							has_trait = trait_hive_mind
						}
					}
				}
			}
			withdraw_ip = {
				ip_target = realdev_assembly_speed
				ip_withdrawn = value:realdev_planet_target_calc|TARGET_VARIABLE|realdev_assembly_speed|
			}
		}
		
		withdraw_ip = {
			ip_target = realdev_production_boost
			ip_withdrawn = value:realdev_planet_target_calc_no_adjustment|TARGET_VARIABLE|realdev_production_boost|
		}
		#Check if we have any megas building before giving it IP
		if = {
			limit = {

			}

				ip_target = realdev_mega_build_speed
				ip_withdrawn = value:realdev_country_target_calc|TARGET_VARIABLE|realdev_mega_build_speed|MULT_VARIABLE|realdev_building_megas|
			}
		}
		if = {
			limit = {
				check_variable = { which = realdev_total_ip value > 0 }
			}
			log = "Country [this.GetName] has Spare IP [this.realdev_total_ip]"

				}
				realdev_assembly_speed_target = {
					log = "Country [this.GetName] is sending Spare IP to Assembly Speed"
					add_variable = { which = realdev_assembly_speed value = realdev_total_ip }
					set_variable = { which = realdev_total_ip value = 0 }
				}
				realdev_production_boost_target = {
					log = "Country [this.GetName] is sending Spare IP to Production Boost"
					add_variable = { which = realdev_production_boost value = realdev_total_ip }
					set_variable = { which = realdev_total_ip value = 0 }
				}
				realdev_mega_build_speed_target = {
					log = "Country [this.GetName] is sending Spare IP to Megastructure Build Speed"
					add_variable = { which = realdev_mega_build_speed value = realdev_total_ip }
					set_variable = { which = realdev_total_ip value = 0 }
				}
				realdev_even_distribution_target = {
					divide_variable = { which = realdev_total_ip value = value:realdev_even_distribution_targets }
					add_variable = { which = realdev_planet_build_speed value = realdev_total_ip }
					add_variable = { which = realdev_ship_build_speed value = realdev_total_ip }
					if = {
						limit = {
							any_owned_species = {
								is_sapient = yes
								OR = {
									is_robotic = yes
									has_trait = trait_plantoid_budding
									PREV = { has_origin = origin_clone_army }
									PREV = { has_civic = civic_permanent_employment }
									has_trait = trait_clone_soldier_infertile
									has_trait = trait_clone_soldier_infertile_full_potential
									AND = {
										is_organic_species = yes
										PREV = { is_hive_empire = yes }
										has_trait = trait_hive_mind
									}
								}
							}
						}
						add_variable = { which = realdev_assembly_speed value = realdev_total_ip }
					}
					add_variable = { which = realdev_production_boost value = realdev_total_ip }
					if = {
						limit = {
							country_has_megastructures_building = yes
						}
						add_variable = { which = realdev_mega_build_speed value = realdev_total_ip }
					}
					set_variable = { which = realdev_total_ip value = 0 }
					
				}
			}
		}
		#If megastructure build speed is 0, set it to 1 so it doesn't break stuff
		if = {
			limit = {
				country_has_megastructures_building = yes
				check_variable = { which = realdev_mega_build_speed value < 1 }
			}
			set_variable = { which = realdev_mega_build_speed value = 1 }
		}
		
		#Set display variables
		set_variable = { which = realdev_planet_build_speed_display value = realdev_planet_build_speed }
		set_variable = { which = realdev_ship_build_speed_display value = realdev_ship_build_speed }

		set_variable = { which = realdev_production_boost_display value = realdev_production_boost }
		set_variable = { which = realdev_mega_build_speed_display value = realdev_mega_build_speed }
		
	}	
}

#Event to place the all modifiers at the start of the month. Math is done before, so as to not interfere with Start of Month stuff.
country_event = {
	id = realdev.5
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = no_realdev }
	}
	
	immediate = {
		#Planet Build Speed
		decay_modifier_update = {
			decay_value = 0.9
			realdev_modifier = realdev_planet_build_speed
			realdev_variable = realdev_planet_build_speed
			realdev_conversion_ratio = @realdev_planet_build_speed_conversion
			division_value = trigger:num_owned_colonies
		}
		#Ship Build Speed
		decay_modifier_update = {
			decay_value = 0.85
			realdev_modifier = realdev_ship_build_speed
			realdev_variable = realdev_ship_build_speed
			realdev_conversion_ratio = @realdev_ship_build_speed_conversion
			division_value = realdev_shipyard_count
		}
		#Planet Assembly Speed
		decay_modifier_update = {
			decay_value = 0.9
			realdev_modifier = realdev_assembly_speed
			realdev_variable = realdev_assembly_speed
			realdev_conversion_ratio = @realdev_assembly_speed_conversion
			division_value = trigger:num_owned_colonies
		}
		#Planet Production Boost
		decay_modifier_no_adjustment_update = {
			decay_value = 0.9
			realdev_modifier = realdev_production_boost
			realdev_variable = realdev_production_boost
			realdev_conversion_ratio = @realdev_production_boost_conversion
			division_value = trigger:num_owned_colonies
		}
		#Mega Build Speed
		decay_modifier_update = {
			decay_value = 0.75
			realdev_modifier = realdev_mega_build_speed
			realdev_variable = realdev_mega_build_speed
			realdev_conversion_ratio = @realdev_mega_build_speed_conversion
			division_value = realdev_building_megas
		}
		reset_realdev_variables = yes
		
		
		#Send a message if IP went down by too much.
		if = {

#Event to handle leaders
leader_event = {
	id = realdev.71
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = no_realdev }
		exists = this
	}
	
	immediate = {
		export_trigger_value_to_variable = {
			trigger = has_total_skill
			variable = realdev_leader_level
		}
		set_variable = { which = realdev_industry value = 0 }
		set_variable = { which = realdev_ship_build_speed value = 0 }
		if = {
			limit = {
				has_trait = leader_trait_architectural_interest
			}
			change_variable = { which = realdev_industry value = 10 }
		}
		if = {
			limit = {
				has_trait = leader_trait_bureaucrat
			}
			change_variable = { which = realdev_industry value = 10 }
		}
		if = {
			limit = {
				has_trait = leader_trait_retired_fleet_officer
			}
			change_variable = { which = realdev_ship_build_speed value = 5 }
		}
		if = {
			limit = {
				leader_class = governor
			}
			multiply_variable = { which = realdev_leader_level value = 2 }
			multiply_variable = { which = realdev_industry value = 2 }
			multiply_variable = { which = realdev_ship_build_speed value = 2 }
		}
		#Councillors
		else_if = {
			limit = {
				is_councilor = yes
			}
			export_trigger_value_to_variable = {
				trigger = has_total_skill
				variable = realdev_leader_level
			}
			divide_variable = { which = realdev_leader_level value = 2 }
			ceiling_variable = realdev_leader_level
		}
	}
}
