namespace = autofab

event = {
	id = autofab.1
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		NOT = { has_global_flag = no_autofab }
	}
	
	immediate = {
		every_country = {
			limit = {
				NOT = {
					has_global_flag = autofab_start
				}
			}
			set_global_flag = autofab_start
			country_event = { id = autofab.2 days = 1 }
		}
	}
	after = {
	}
}
