realdev_country_tooltip = {
	potential = { }
	allow = { hidden_trigger = { always = yes } }
	effect = { custom_tooltip = "realdev_country_tooltip" }
}

realdev_select_all_targets = {
	potential = {
		NOT = {
			has_country_flag = realdev_planet_build_speed_change_target
			has_country_flag = realdev_ship_build_speed_change_target
			has_country_flag = realdev_assembly_speed_change_target
			has_country_flag = realdev_production_boost_change_target
			has_country_flag = realdev_mega_build_speed_change_target
		}
	}
	allow = { hidden_trigger = { always = yes } }
	effect = {
		custom_tooltip = "realdev_select_all_targets"
		hidden_effect = {
			set_country_flag = realdev_planet_build_speed_change_target
			set_country_flag = realdev_ship_build_speed_change_target
			set_country_flag = realdev_assembly_speed_change_target
			set_country_flag = realdev_production_boost_change_target
			set_country_flag = realdev_mega_build_speed_change_target
		}
	}
}
realdev_unselect_all_targets = {

###Increase Targets
realdev_increase_target_100 = {
	potential = { }
	allow = { hidden_trigger = { always = yes } }
	effect = {
		custom_tooltip = "realdev_increase_target_100"
		hidden_effect = {
			if = {
				limit = {
					has_country_flag = realdev_planet_build_speed_change_target
				}
				add_variable = { which = realdev_planet_build_speed_target value = 100 }
			}
			if = {
				limit = {
					has_country_flag = realdev_ship_build_speed_change_target
				}
				add_variable = { which = realdev_ship_build_speed_target value = 100 }
			}
			if = {
				limit = {
					has_country_flag = realdev_assembly_speed_change_target
				}
				add_variable = { which = realdev_assembly_speed_target value = 100 }
			}
			if = {
				limit = {
					has_country_flag = realdev_production_boost_change_target
				}
				add_variable = { which = realdev_production_boost_target value = 100 }
			}
			if = {
				limit = {
					has_country_flag = realdev_mega_build_speed_change_target
				}
				add_variable = { which = realdev_mega_build_speed_target value = 100 }
			}
		}
	}
}

		custom_tooltip = "realdev_country_tooltip"
