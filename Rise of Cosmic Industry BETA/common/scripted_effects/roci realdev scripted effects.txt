decay_modifier_update = {
	multiply_variable = { which = $realdev_variable$_perm value = $decay_value$ } #Loses X% effectiveness
	
	divide_variable = { which = $realdev_variable$ value = $realdev_conversion_ratio$ } #Divide IP by conversion ratio...
	change_variable = { which = $realdev_variable$_perm value = $realdev_variable$ } #...then add to perm variable
	
	set_variable = { which = $realdev_variable$_perm_display value = $realdev_variable$_perm }
	subtract_variable = { which = $realdev_variable$_perm_display value = realdev_realism_adjustment }
	
	add_modifier = {
		modifier = $realdev_modifier$
		multiplier = value:realdev_adjustment_applicator|IP_VARIABLE|$realdev_variable$_perm|DIVISION|$division_value$|
		days = 30
	}
}

withdraw_ip = {
	#log = "withdrawing IP for \\[this.GetName\\]"
	#set_variable = { which = test value = $ip_withdrawn$ }
	#log = "IP to withdraw \\[this.test\\] to $ip_target$"
	set_variable = { which = $ip_target$_target_display value = $ip_withdrawn$ }
	#Subtract from the target how much already exists
	set_variable = { which = ip_to_withdraw value = $ip_withdrawn$ }
	subtract_variable = { which = ip_to_withdraw value = $ip_target$ }
	if = {
		limit = {
			check_variable = { which = realdev_total_ip value >= ip_to_withdraw }
		}
		subtract_variable = { which = realdev_total_ip value = ip_to_withdraw }
		add_variable = { which = $ip_target$ value = ip_to_withdraw }
	}
	else = {
		add_variable = { which = $ip_target$ value = realdev_total_ip }
		set_variable = { which = realdev_total_ip value = 0 }
	}
	round_variable = $ip_target$
}
